4. Aging Fleet: An Honest Within-Carrier Test
▶ SQL Query Statement
-- holds carrier fixed: each airline's own pre-2000 fleet vs its own 2000+ fleet WITH carrier_age_buckets AS ( SELECT f.carrier, CASE WHEN p.year < 2000 THEN 'pre_2000' ELSE 'y2000_plus' END AS age_bucket, f.dep_delay FROM flights f INNER JOIN planes p ON f.tailnum = p.tailnum WHERE p.year IS NOT NULL AND f.dep_delay IS NOT NULL ), carrier_summary AS ( SELECT carrier, age_bucket, COUNT(*) AS n_flights, ROUND(AVG(dep_delay),2) AS avg_delay FROM carrier_age_buckets GROUP BY carrier, age_bucket HAVING COUNT(*) >= 500 ) SELECT a.name AS airline_name, old.n_flights AS pre_2000_flights, old.avg_delay AS pre_2000_avg_delay, new.n_flights AS y2000_plus_flights, new.avg_delay AS y2000_plus_avg_delay, ROUND(old.avg_delay - new.avg_delay, 2) AS age_penalty_minutes FROM carrier_summary old JOIN carrier_summary new ON old.carrier = new.carrier INNER JOIN airlines a ON old.carrier = a.carrier WHERE old.age_bucket = 'pre_2000' AND new.age_bucket = 'y2000_plus' ORDER BY age_penalty_minutes DESC;

Window Function Rank BreakdownQuery ID: q5_window_rank
5. Route Efficiency Leaderboard: Real Competition Only
▶ SQL Query Statement
WITH RoutePerformance AS ( SELECT f.origin, f.dest, a.name AS airline_name, COUNT(f.flight_id) AS total_flights, ROUND(AVG(CASE WHEN f.dep_delay > 0 THEN f.dep_delay ELSE 0 END), 1) AS avg_delay_minutes FROM flights f INNER JOIN airlines a ON f.carrier = a.carrier GROUP BY f.origin, f.dest, a.name HAVING total_flights > 1000 ), ranked AS ( SELECT *, RANK() OVER (PARTITION BY origin, dest ORDER BY avg_delay_minutes ASC) AS route_efficiency_rank, COUNT(*) OVER (PARTITION BY origin, dest) AS carriers_on_route FROM RoutePerformance ), -- margin = gap between the winner and runner-up on each route, used to rank ROUTES, not rows margins AS ( SELECT origin, dest, MAX(CASE WHEN route_efficiency_rank = 1 THEN avg_delay_minutes END) AS winner_delay, MIN(CASE WHEN route_efficiency_rank = 2 THEN avg_delay_minutes END) AS runnerup_delay FROM ranked WHERE carriers_on_route > 1 AND route_efficiency_rank <= 2 GROUP BY origin, dest ) SELECT r.origin, r.dest, r.airline_name, r.total_flights, r.avg_delay_minutes, r.route_efficiency_rank, ROUND(m.runnerup_delay - m.winner_delay, 1) AS margin FROM ranked r JOIN margins m ON r.origin = m.origin AND r.dest = m.dest WHERE r.route_efficiency_rank <= 2 ORDER BY margin DESC, r.origin, r.dest, r.route_efficiency_rank LIMIT 12;

v_daily_airport_performance

Isolates weather-driven delay from carrier scheduling
▶ View Definition
CREATE OR REPLACE VIEW v_daily_airport_performance AS WITH daily_weather AS ( SELECT origin, year, month, day, AVG(temp) avg_temp, AVG(wind_speed) avg_wind FROM weather GROUP BY origin, year, month, day ) SELECT f.origin, ap.name, f.year, f.month, f.day, COUNT(*) total_departures, ROUND(AVG(COALESCE(f.dep_delay,0)),2) avg_dep_delay, ROUND(MAX(dw.avg_temp),1) avg_temp_f, ROUND(MAX(dw.avg_wind),1) avg_wind_mph FROM flights f INNER JOIN airports ap ON f.origin=ap.faa LEFT JOIN daily_weather dw ON f.origin=dw.origin AND f.year=dw.year AND f.month=dw.month AND f.day=dw.day GROUP BY f.origin, ap.name, f.year, f.month, f.day;

Question 6 · Runtime Benchmark

Subquery vs. CTE

Which flights ran later than typical for their own route? Answered two ways, benchmarked on the real submission database.

Not comparable at full scale:
  • CTE: finished the full 336,776-row table in under a second.
  • Subquery: did not finish inside a 3,000-second (50-min) timeout — abandoned.
  • Instead: subquery run on a 5,000-row slice, twice, and extrapolated.
Version A — Correlated Subquery
SELECT f.flight_id, f.origin, f.dest, f.carrier, f.dep_delay FROM flights f WHERE f.dep_delay IS NOT NULL AND f.dep_delay > ( SELECT AVG(f2.dep_delay) FROM flights f2 WHERE f2.origin = f.origin AND f2.dest = f.dest AND f2.dep_delay IS NOT NULL );
Version B — CTE Pre-aggregated
WITH route_avg AS ( SELECT origin, dest, AVG(dep_delay) AS route_avg_delay FROM flights WHERE dep_delay IS NOT NULL GROUP BY origin, dest ) SELECT f.flight_id, f.origin, f.dest, f.carrier, f.dep_delay FROM flights f JOIN route_avg ra ON f.origin = ra.origin AND f.dest = ra.dest WHERE f.dep_delay IS NOT NULL AND f.dep_delay > ra.route_avg_delay;

  • MySQL 8 has no result cache for this correlated subquery — it recomputes AVG(dep_delay) per row, not per route.
  • The CTE computes all 223 route averages once, then does cheap indexed lookups.
  • This doesn't generalize to all subqueries — only ones re-aggregating a value repeated across many rows.
Bottom line: use the CTE. Same hardware, ~1,716× faster, and the subquery may not finish at all at real table sizes.
